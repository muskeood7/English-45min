                <button class="btn" style="width:auto; padding:6px 12px; font-size:12px;" onclick="generateDailyContent()">🤖 生成新内容</button>

            <hr style="border:none; border-top:1px solid #E5E7EB; margin: 16px 0;">

            <!-- 15分钟：听力训练 -->
            <div style="margin-bottom: 24px;">

            </div>

            <div style="margin-bottom: 24px;">

                <button class="btn btn-outline" onclick="speakText('reading-text')">🔊 朗读阅读文章</button>
            </div>

    <script>
        // 全局状态管理（借助 localStorage 实现本地数据持久化）
        let userLevel = localStorage.getItem('user_level') || null;
        let streak = parseInt(localStorage.getItem('streak') || '0');
        let apiKey = localStorage.getItem('api_key') || '';
        let currentQuestion = 0;
        let score = 0;

        // 评测题目
        const questions = [
            {
                q: "1. Select the correct word: 'She _____ to work by subway every day.'",
                opts: ["A. go", "B. goes", "C. going", "D. gone"],
                ans: 1
            },
            {
                q: "2. What is the closest meaning to 'Crucial'?",
                opts: ["A. Minor", "B. Extremely Important", "C. Easy", "D. Slow"],
                ans: 1
            },
            {
                q: "3. Complete the sentence: 'If I had known about the traffic, I _____ earlier.'",
                opts: ["A. will leave", "B. would have left", "C. left", "D. leave"],
                ans: 1
            }
        ];

        // 本地内置备份内容（无 API 时自动调用）
        const contentDB = {
            "L1": {
                listening: "Hello! My name is Amy. I live in a quiet city. Every morning, I like to drink green tea and read short stories.",
                reading: "Tom has a friendly dog named Max. Max loves running in the park and playing fetch with a small ball.",
                shadowing: "Consistency is the key to acquiring any new skills."
            },
            "L2": {
                listening: "Effective communication is not just about using complex vocabulary, but also about active listening and clear expression.",
                reading: "Forming a daily study habit takes dedication. Spending 45 minutes every day reading and listening helps train your mind in English.",
                shadowing: "Small daily improvements over time lead to stunning long-term results."
            },
            "L3": {
                listening: "Technological advancements have radically transformed how modern learners access educational materials globally.",
                reading: "Language acquisition thrives on comprehensive input. Engaging with content slightly above your current proficiency accelerates intuitive understanding.",
                shadowing: "Mastering a second language unveils a completely original perspective on the world."
            }
        };

        function initApp() {
            document.getElementById('streak-text').innerText = `连续打卡：${streak} 天`;
            if (apiKey) document.getElementById('api-key-input').value = apiKey;

            if (userLevel) {
                document.getElementById('user-level-tag').innerText = `当前等级：${userLevel}`;
                document.getElementById('assessment-card').classList.add('hidden');
                document.getElementById('learning-card').classList.remove('hidden');
                generateDailyContent();
            } else {
                startAssessment();
            }
        }

        function saveApiKey() {
            apiKey = document.getElementById('api-key-input').value.trim();
            localStorage.setItem('api_key', apiKey);
            alert('API 密钥配置已保存！');
        }

        function startAssessment() {
            currentQuestion = 0;
            score = 0;
            document.getElementById('assessment-card').classList.remove('hidden');
            document.getElementById('learning-card').classList.add('hidden');
            showQuestion();
        }

        function showQuestion() {
            if (currentQuestion >= questions.length) {
                finishAssessment();
                return;
            }
            let qData = questions[currentQuestion];
            document.getElementById('quiz-q').innerText = qData.q;
            let html = '';
            qData.opts.forEach((opt, idx) => {
                html += `<button class="quiz-opt" onclick="answerQuiz(${idx})">${opt}</button>`;
            });
            document.getElementById('quiz-options').innerHTML = html;
        }

        function answerQuiz(idx) {
            if (idx === questions[currentQuestion].ans) score++;
            currentQuestion++;
            showQuestion();
        }

        function finishAssessment() {
            if (score <= 1) userLevel = 'L1 (入门级)';
            else if (score === 2) userLevel = 'L2 (基础级)';
            else userLevel = 'L3 (进阶级)';

            localStorage.setItem('user_level', userLevel);
            document.getElementById('user-level-tag').innerText = `当前等级：${userLevel}`;
            alert(`🎉 测评完成！评估结果为：${userLevel}。系统已为您配置适配的推送内容！`);
            
            document.getElementById('assessment-card').classList.add('hidden');
            document.getElementById('learning-card').classList.remove('hidden');
            generateDailyContent();
        }

        async function generateDailyContent() {
            let lvlKey = userLevel.startsWith('L1') ? 'L1' : (userLevel.startsWith('L2') ? 'L2' : 'L3');

            // 如果配置了 API Key 则调用在线大模型 API
            if (apiKey) {
                document.getElementById('listening-text').innerText = "🤖 AI 正在生成专属听力文本...";
                document.getElementById('reading-text').innerText = "🤖 AI 正在生成专属阅读文章...";
                document.getElementById('shadowing-text').innerText = "🤖 AI 正在生成专属跟读句子...";
                try {
                    const response = await fetch("https://api.openai.com/v1/chat/completions", {
                        method: "POST",
                        headers: {
                            "Content-Type": "application/json",
                            "Authorization": `Bearer ${apiKey}`
                        },
                        body: JSON.stringify({
                            model: "gpt-4o-mini",
                            messages: [{
                                role: "user",
                                content: `请为英语 ${userLevel} 等级初学者生成 1 组每日学习卡片，包含：listening (80词听力短文), reading (120词分级阅读), shadowing (1句跟读金句)。请用纯 JSON 格式返回，格式如下: {"listening":"...","reading":"...","shadowing":"..."}`
                            }]
                        })
                    });
                    const resData = await response.json();
                    const content = JSON.parse(resData.choices[0].message.content);
                    document.getElementById('listening-text').innerText = content.listening;
                    document.getElementById('reading-text').innerText = content.reading;
                    document.getElementById('shadowing-text').innerText = content.shadowing;
                    return;
                } catch (e) {
                    console.warn("API 调用失败或未响应，将自动使用本地精选库：", e);
                }
            }

            // 未配置 API Key 或 API 异常时，使用本地默认数据库
            let data = contentDB[lvlKey];
            document.getElementById('listening-text').innerText = data.listening;
            document.getElementById('reading-text').innerText = data.reading;
            document.getElementById('shadowing-text').innerText = data.shadowing;
        }

        // Web Speech API 语音朗读控制
        function speakText(elementId) {
            let text = document.getElementById(elementId).innerText;
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel(); // 停止当前所有朗读队列
                let msg = new SpeechSynthesisUtterance(text);
                msg.lang = 'en-US';
                msg.rate = 0.9; // 调整为适合听力练习的 0.9x 语速
                window.speechSynthesis.speak(msg);
            } else {
                alert('当前浏览器环境不支持原声发音，推荐在 iOS Safari 或 Chrome 中打开使用。');
            }
        }

        // 完成打卡
        function doCheckIn() {
            streak++;
            localStorage.setItem('streak', streak);
            document.getElementById('streak-text').innerText = `连续打卡：${streak} 天`;
            alert(`🎉 打卡成功！你已完成今日 45 分钟学习。连续打卡天数达：${streak} 天，继续坚持！`);
        }

        // 启动应用
        initApp();
    </script>
